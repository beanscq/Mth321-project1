\section{What the problem is}
% =====================================================================

\begin{frame}{Three chemical species --- how much of each is there?}
  \begin{block}{A three-species reaction system (the Robertson problem)}
    \begin{align*}
      y_1' &= -0.04\,y_1 + 10^{4}\,y_2 y_3 \\
      y_2' &= \phantom{-}0.04\,y_1 - 10^{4}\,y_2 y_3 - 3\times 10^{7}\,y_2^{2} \\
      y_3' &= \phantom{-0.04\,y_1 - 10^{4}\,y_2 y_3 +{}} 3\times 10^{7}\,y_2^{2}
    \end{align*}
    \[
      y(0)=(1,\,0,\,0)^{\mathsf T},\qquad t\in[0,\,40]
    \]
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \textbf{In one sentence}\\[0.3em]
      Species A, B and C interconvert, and at the start everything is A.
      We want to know how much of each is present at
      \alert{every instant} of the $0$--$40$ second window.
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{What we have to hand in}\\[0.3em]
      \begin{itemize}
        \item three concentration curves $y_1(t),y_2(t),y_3(t)$
        \item one number: \alert{$y_2(40)$}
        \item plus an argument: \emph{why you should believe it}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{What the answer looks like: one sharp spike, two slow curves}
  \begin{center}
    \includegraphics[width=0.87\linewidth]{F2_reference_solution.pdf}
  \end{center}
  \vspace{-0.35em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Start with B (the middle curve)}
        \footnotesize
        B shoots up from zero, peaks at $3.65\times10^{-5}$ at
        $t=4.42\times10^{-3}$ s, then decays by 5 orders of magnitude,
        down to $9.18553\times10^{-6}$ at $t=40$ s.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Now A and C}
        \footnotesize
        A drifts down from $1$ to $0.7158$; C drifts up from $0$ to $0.2842$.
        Both of these move on a scale of \alert{tens of seconds} ---
        a completely different world from B.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \begin{center}\footnotesize \fig{F2}: the reference solution, time on a logarithmic axis.\end{center}
\end{frame}

\section{Why it is hard}
% =====================================================================

\begin{frame}{Why it is hard: B is a clock that ticks in microseconds}
  \begin{center}
    \includegraphics[width=0.96\linewidth]{F13_two_clocks.pdf}
  \end{center}
  \vspace{-0.3em}
  \begin{columns}[T]
    \begin{column}{0.30\textwidth}
      \footnotesize \fig{F13}: concentrations (left) and their clocks
      (right), i.e.\ characteristic times $1/|\mathrm{Re}\,\lambda|$.
    \end{column}
    \begin{column}{0.66\textwidth}
      \footnotesize
      B is extremely reactive: the moment it forms it is consumed within
      microseconds, so its concentration never leaves the $10^{-5}$ range.
      The system therefore carries \textbf{two clocks of wildly different
      rate} --- only $1.7\times$ apart at the start, but
      \alert{$1.58\times10^{5}\times$} apart by $t=40$.
      On the next slide we put a number on that ratio and give it a name.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Why it is hard (1): how different are the two clocks? --- stiffness}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \includegraphics[width=\linewidth]{F6_stiffness_ratio.pdf}
      \begin{center}\footnotesize \fig{F6}: $S(t)=\max|\mathrm{Re}\,\lambda|/
      \min|\mathrm{Re}\,\lambda|$, counting only the two non-zero eigenvalues\end{center}
    \end{column}
    \begin{column}{0.54\textwidth}
      \begin{block}{A basic concept first: what is stiffness?}
        \footnotesize
        At any time $t$, the eigenvalues $\lambda$ of $J(t)$ give the time
        scales $\tau=1/|\mathrm{Re}\,\lambda|$ of each mode, so
        \[
          S(t)=\frac{\max|\mathrm{Re}\,\lambda|}{\min|\mathrm{Re}\,\lambda|}
        \]
        is the ratio of the fast and slow clocks.  When $S\gg1$ the problem
        is \alert{stiff}.
      \end{block}
      {\footnotesize
      \setlength{\tabcolsep}{6pt}%
      \begin{tabular}{@{}lrr@{}}
        \toprule
        $t$ & our $S$ & reference value \\
        \midrule
        $10^{-4}$ & $3.0097\times10^{3}$ & $3.0\times10^{3}$ \\
        $10^{-2}$ & $5.4221\times10^{3}$ & $5.4\times10^{3}$ \\
        $40$      & $1.5840\times10^{5}$ & $1.58\times10^{5}$ \\
        \bottomrule
      \end{tabular}}
      \vspace{0.2em}

      \begin{alertblock}{The stiffest moment is $t=40$, not $t=0$}
        \footnotesize
        The three points agree to $0.25\%$--$0.41\%$; at $t=10^{-8}$ the ratio
        is only $1.67$, while at $t=40$ it is $1.58\times10^{5}$.  So the
        step-size limit must use the $\lambda$ at $t=40$, not at the start.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Why it is hard (2): explicit methods have a short stability interval}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \includegraphics[width=\linewidth]{F4_stability_regions.pdf}
      \begin{center}\footnotesize \fig{F4}: absolute stability regions of the three methods\end{center}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Another basic concept: absolute stability region}
        \footnotesize
        Write one step as $y_{n+1}=R(z)\,y_n$ with $z=h\lambda$.  The method
        is \alert{absolutely stable} when $|R(z)|\le1$ (errors are not
        amplified).  The set of such $z$ is the stability region.
        \begin{itemize}
          \setlength{\itemsep}{0.05em}
          \item For explicit methods $R$ is a \emph{polynomial}, so the region
                is \alert{bounded}.
          \item For implicit Euler $R(z)=1/(1-z)$ and $|R|\le1$ everywhere in
                the left half-plane.
        \end{itemize}
      \end{block}
      {\small
      \setlength{\tabcolsep}{5pt}%
      \begin{tabular}{@{}llr@{}}
        \toprule
        method & real-axis interval & order \\
        \midrule
        Explicit Euler & $[-2.0000,\,0]$ & 1 \\
        RK4            & $[-2.7853,\,0]$ & 4 \\
        Implicit Euler & $(-\infty,\,0]$ & 1 \\
        \bottomrule
      \end{tabular}}
      \vspace{0.3em}

      Substituting $z=h\lambda$ gives
      $h_{\rm EE}<5.895\times10^{-4}$, $h_{\rm RK4}<8.209\times10^{-4}$;
      implicit Euler has \alert{no limit} in this direction.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Consequence: the upper bound nails the cost}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{alertblock}{A hard lower bound on cost}
        Use $|\lambda|_{\max}=3392.79$ (at $t=40$) in $h<2/|\lambda|$:
        \[
          h < 5.9\times10^{-4}\ \mathrm{s}
        \]
        Crossing $[0,40]$ therefore needs at least
        \[
          N \;\ge\; \frac{40}{5.9\times10^{-4}} \;\approx\; 6.8\times10^{4}
        \]
      \end{alertblock}
      \footnotesize
      This is a \emph{lower bound}, not a recommendation: explicit methods
      cannot even fill this many steps --- the measured largest stable step is
      $8\%$ smaller (Conclusion 2).
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{The awkward part: the limit is at the end but governs all}
        \footnotesize
        $|\lambda|_{\max}$ is the maximum over the \emph{whole interval}, and
        it occurs at $t=40$, not at the start.  Long after B has settled and
        nothing in the system is changing quickly, the restriction is
        \alert{still there}.  The fast process \emph{pins the global step size}.
      \end{block}
      \begin{block}{Why we cannot ``only be careful at the beginning''}
        \footnotesize
        Stability is a \emph{cumulative} test: once a step crosses the
        boundary, $|R(z)|>1$ and the error is amplified at every subsequent
        step.  Raising $h$ from $5.4159\times10^{-4}$ by only $15\%$ already
        produces a concentration of \alert{$-182$} at $t=32.9$.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{How we did it}
% =====================================================================

\begin{frame}{How we did it: build a ruler, then measure three methods}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Step one: build a ruler}
        \footnotesize
        Compute a \alert{reference solution} with a high-accuracy solver, then
        confirm it by three independent routes, agreeing to \alert{10 digits}.
        \emph{Without that ruler the words ``how accurate'' mean nothing.}
      \end{block}
      \begin{block}{Step two: three methods, and check their orders first}
        \footnotesize
        \textbf{Explicit Euler} (1)\quad \textbf{RK4} (4)\quad
        \textbf{Implicit Euler} (1).  All written from scratch; then we measure
        their \emph{observed orders} to verify that the formulas are correct.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Step three: put every method through four checks}
        \footnotesize
        Conservation / non-negativity / accuracy / endpoint.
        The four must be read \alert{separately} --- they do \emph{not} imply
        one another, and the biggest trap in this problem is hidden in one of
        the combinations.  (How each is measured is on the next slide.)
      \end{block}
      \begin{exampleblock}{Division of labour}
        \footnotesize
        Ruler (independent) $\to$ methods (ours) $\to$ checks (separate) $\to$
        conclusions (each directly verifiable from the data above).
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Step one: build a ruler we can trust}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Three mutually independent confirmations (\tab{T4})}\\[0.3em]
      \begin{itemize}
        \item \textbf{Tolerance repetition}: rerun the same solver at tighter
              tolerances and see at which digit the numbers stop moving
        \item \textbf{Cross-validation}: compare against SciPy's \code{BDF}
              and \code{LSODA}
        \item \textbf{Structural identity}: $\operatorname{m2}(J)$ must equal
              $\det(B_{\rm red})$
      \end{itemize}
      \vspace{0.2em}

      \footnotesize The three routes fail in different ways (step size,
      implementation, algebra), so agreement among them is meaningful.
    \end{column}
    \begin{column}{0.44\textwidth}
      {\small
      \begin{tabular}{@{}lr@{}}
        \toprule
        Confirmation route & agreeing digits \\
        \midrule
        tolerance repetition & 10.76 \\
        vs \code{BDF}        & 10.70 \\
        vs \code{LSODA}      & 10.98 \\
        \midrule
        \textbf{lower bound used} & \textbf{10.76} \\
        \bottomrule
      \end{tabular}}
      \vspace{0.4em}

      \footnotesize Largest componentwise deviation $2.089\times10^{-12}$.
    \end{column}
  \end{columns}

  \vspace{0.4em}
  \begin{block}{This is the yardstick for everything that follows}
    The reference gives $y_2(40)=9.185535\times10^{-6}$, quotable to 10 digits.
    Every error quoted later is an \alert{pointwise comparison against it}.
  \end{block}
\end{frame}

\begin{frame}{Step two: what the three methods actually are}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Explicit Euler (1st order)}
        \footnotesize
        \[
          y_{n+1}=y_n+h\,f(t_n,y_n)
        \]
        The right-hand side uses only the known $y_n$, so one step costs
        one evaluation of $f$.  The amplification factor is $R(z)=1+z$.
      \end{block}
      \begin{block}{RK4 (4th order)}
        \footnotesize
        \[
          y_{n+1}=y_n+\tfrac{h}{6}\bigl(k_1+2k_2+2k_3+k_4\bigr)
        \]
        A weighted average of four slopes, four evaluations of $f$ per step:
        \alert{four times the work for fourth-order accuracy}.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Implicit Euler (1st order)}
        \footnotesize
        \[
          y_{n+1}=y_n+h\,f(t_{n+1},y_{n+1})
        \]
        The unknown appears on \emph{both} sides: each step solves
        \[
          G(u)=u-y_n-hf(u)=0
        \]
        with Newton's method, requiring $J$ and one linear solve per step.
      \end{block}
      \begin{alertblock}{The difference is only two things}
        \footnotesize
        (1) how many $f$ evaluations per step (work); and
        (2) \emph{whether the unknown future value is used} (stability).
        RK4 is good at (1) and bad at (2); implicit Euler is the opposite.
        All later conclusions are intersections of these two lines.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Step two (cont.): first check that the observed orders are right}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \includegraphics[width=0.98\linewidth]{F8_convergence.pdf}
      \begin{center}\footnotesize \fig{F8}: error vs $h$ on a log-log scale\end{center}
      \vspace{0.2em}
      \footnotesize
      \textbf{How we measure it}: repeatedly halve $h$, measure the error, and
      read the slope on the log-log plot.
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{block}{Basics: on a log-log plot the slope is the order}
        \footnotesize
        If the error $\approx C h^{p}$, then $\log(\text{err})$ vs $\log h$ is a
        straight line of slope $p$.  So the order is something we can
        \alert{measure}, not just trust from a formula.
      \end{block}
      {\footnotesize
      \setlength{\tabcolsep}{5pt}%
      \begin{tabular}{@{}lcrr@{}}
        \toprule
        method & nominal & fitted slope & $R^2$ \\
        \midrule
        EE  & 1 & $1.079{\pm}0.033$ & 0.9982 \\
        RK4 & 4 & $4.268{\pm}0.074$ & 0.9994 \\
        IE  & 1 & $0.99996{\pm}1.2{\times}10^{-5}$ & 1.0000 \\
        \bottomrule
      \end{tabular}}
      \vspace{0.25em}

      \footnotesize
      All three are first interpolated by cubic Hermite onto the same
      201-point grid before comparison.  If the interpolation were only
      $O(h^4)$, the observed RK4 order would be capped at 2; the measured
      $4.27$ shows it \alert{is not}.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Step three: what the four checks measure, and why none can be skipped}
  \begin{columns}[T]
    \begin{column}{0.487\textwidth}
      \checkcard{cBlue}{\ckbadge{cBlue}{1}~Conservation}{%
        \textbf{measured}: $\max_t\,\lvert y_1+y_2+y_3-1\rvert$, every run\\[0.2em]
        \textbf{why separately}: in this problem the sum is
        \alert{structurally exact}, so this check has
        \alert{no discriminating power} (see Conclusion 1)}
      \vspace{0.4em}
      \checkcard{cGreen}{\ckbadge{cGreen}{2}~Non-negativity}{%
        \textbf{measured}: $\min_i y_i$, every run\\[0.2em]
        \textbf{why separately}: concentrations are physical quantities;
        $y_i<0$ means the solution already has \emph{no meaning}, even if
        conservation looks fine}
    \end{column}
    \begin{column}{0.487\textwidth}
      \checkcard{cRed}{\ckbadge{cRed}{3}~Accuracy}{%
        \textbf{measured}: pointwise maximum error against the reference\\[0.2em]
        \textbf{why separately}: the only check that really measures
        ``accuracy'' --- and the only one that can detect
        ``stable but wrong''}
      \vspace{0.4em}
      \checkcard{cOrange}{\ckbadge{cOrange}{4}~Endpoint}{%
        \textbf{measured}: $y_2(40)$ against the reference value\\[0.2em]
        \textbf{why separately}: it is the requested number, but it reflects
        one point only and \alert{cannot represent the whole interval}}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \footnotesize
  \ckbadge{cBlue}{1}\ckbadge{cGreen}{2}\ckbadge{cRed}{3}\ pass while
  \ckbadge{cOrange}{4}\ is far off: the four checks are
  \alert{logically independent} --- each can be true or false on its own
  ($2^4=16$ combinations).  The next slide nails this biggest trap with data.
\end{frame}

\section{Four conclusions}
% =====================================================================

\begin{frame}{Conclusion 1: among the four tracks, only one catches the error}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \includegraphics[width=\linewidth]{F9_four_track_diagnostics.pdf}
      \begin{center}\footnotesize \fig{F9}: conservation / non-negativity / error / endpoint, four tracks\end{center}
      \vspace{0.2em}
      \footnotesize
      \textbf{What we did}: for each method, run 22 logarithmically spaced
      step sizes, recording the four quantities simultaneously.  Which track
      fails \emph{first} is immediately visible.
    \end{column}
    \begin{column}{0.57\textwidth}
      \begin{alertblock}{The most glaring run (explicit Euler, $h=3\times10^{-3}$)}
        \footnotesize
        \begin{align*}
          \text{conservation defect} &= 1.1\times10^{-13}\quad(\text{looks perfect})\\
          \min_i y_i &= -921.8\quad(\text{concentration is negative!})\\
          \text{error} &= 1.08\times10^{10}
        \end{align*}
        \alert{Check conservation only and this garbage passes.}
      \end{alertblock}
      \begin{block}{The four tracks are not synchronized}
        \footnotesize
        \begin{itemize}
          \setlength{\itemsep}{0.05em}
          \item the conservation defect is $\sim10^{-14}$ at \emph{every} step
                size --- \alert{no discriminating power at all}
          \item RK4 has a whole window that is stable, non-negative and
                \emph{already inaccurate}
          \item the track that fails first is \emph{not the same} for each
                method: EE loses accuracy and non-negativity together, RK4
                loses accuracy first
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion 1, why?  Why is the conservation defect always $10^{-14}$?}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Step one: add the three equations}
        \footnotesize
        Every term on the right cancels in pairs:
        $-0.04y_1$ with $+0.04y_1$,
        $+10^{4}y_2y_3$ with $-10^{4}y_2y_3$,
        $-3\times10^{7}y_2^{2}$ with $+3\times10^{7}y_2^{2}$.  Hence
        \[
          \frac{d}{dt}\bigl(y_1+y_2+y_3\bigr)\equiv0
          \quad\Longrightarrow\quad S\equiv1
        \]
        $S$ is an \emph{exact invariant} of the system, independent of the
        numerical method.
      \end{block}
      \begin{block}{Step two: all three schemes preserve it automatically}
        \footnotesize
        Explicit Euler: adding the components gives
        $S_{n+1}=S_n+h\sum_i f_i=S_n$ because $\sum_i f_i\equiv0$.  The same
        happens for every RK4 stage and for implicit Euler.  \alert{This holds
        for any step size and any order.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{So it only measures floating-point round-off}
        \footnotesize
        $10^{-14}$ is not ``good accuracy''; it is the magnitude of
        round-off in double precision.  A quantity that is \alert{exactly
        preserved by construction} cannot diagnose accuracy --- it has no
        discriminating power, not because the numerics are good, but because
        \emph{it cannot structurally fail}.
      \end{alertblock}
      \begin{block}{A general rule of thumb}
        \footnotesize
        Whenever you see an ``invariant check'', ask first:
        \emph{is it preserved by the discretisation, or is it affected by
        discretisation error?}  The former only catches implementation bugs;
        it will not catch accuracy problems.
      \end{block}
      \vspace{0.1em}
      \footnotesize
      In this problem the checks that can actually fail are non-negativity
      ($\min_i y_i$), accuracy (pointwise error) and the endpoint ($y_2(40)$).
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion 2: the step-size limit can be predicted in advance, and accurately}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}

      sweep and see where the solution breaks.  These are \alert{two
      independent things}: prediction from linear algebra, measurement from
      real runs.

    \begin{column}{0.44\textwidth}

      about ten percent.
      \begin{block}{Stated honestly}
        \footnotesize

        and the system is strongly non-normal.  So it is a diagnostic, not a

      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion 2, why?  Where does the bound come from}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Idea one: ``freeze'' the nonlinear problem at one instant}
        \footnotesize
        At some time $t$, treat $J(t)$ as \emph{constant}.  Then $y'=Jy$ has
        solution $e^{Jt}$, and one step has amplification factor $R(h\lambda)$.
        Requiring \alert{all} $\lambda$ to satisfy $|R(h\lambda)|\le1$, the
        most restrictive one gives the step-size limit.
      \end{block}
      \begin{block}{Idea two: the two amplification polynomials}
        \footnotesize
        \begin{tabular}{@{}llr@{}}
          \toprule
          method & $R(z)$ & real-axis limit \\
          \midrule
          EE  & $1+z$ & $2.0000$ \\
          RK4 & $1+z+\frac{z^2}{2}+\frac{z^3}{6}+\frac{z^4}{24}$ & $2.7853$ \\
          \bottomrule
        \end{tabular}
        \vspace{0.3em}

        $R$ is a polynomial for explicit methods $\Rightarrow$ $|R|\le1$ only
        on a short real interval $\Rightarrow$
        $h\le\text{limit}\big/|\lambda|_{\max}$.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{alertblock}{Substitute $|\lambda|_{\max}=3392.79$ (at $t=40$)}
        \footnotesize
        \begin{align*}
          h_{\rm EE} &\le \frac{2.0000}{3392.79}=5.895\times10^{-4}\\
          h_{\rm RK4}&\le \frac{2.7853}{3392.79}=8.209\times10^{-4}
        \end{align*}
      \end{alertblock}
      \begin{block}{Why this single bound governs the whole interval}
        \footnotesize
        We use the \emph{global maximum} $|\lambda|_{\max}$.  The bound at any
        other time is never stricter, so one number covers $[0,40]$.
        Using the $t=0$ eigenvalue would overestimate the limit by more than
        $10^5$.
      \end{block}
      \begin{block}{Why the measurement is not exactly equal to it}
        \footnotesize
        $J$ varies and the system is strongly non-normal, so an $8\%$
        deviation is normal.  \alert{Being within ten percent is already
        useful.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion 3, why?  Why ``more stable'' is not ``cheaper''}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{How cost is counted}
        \footnotesize
        We use a single \emph{cost unit}: one right-hand-side evaluation $f$
        counts as $1$, one Jacobian evaluation counts as $1$, and one linear
        solve (LU) counts as $1$; add them up.
        \begin{itemize}
          \setlength{\itemsep}{0.05em}
          \item Explicit Euler: one step $=1$ evaluation of $f$ $\Rightarrow$ $1$
          \item RK4: one step $=4$ evaluations of $f$ $\Rightarrow$ $4$
          \item Implicit Euler: one step $=f+J+\text{solve}$ $\Rightarrow$ $3$
        \end{itemize}
      \end{block}
      \begin{block}{Why this accounting is necessary}
        \footnotesize
        The three methods do different amounts of work per step, so comparing
        ``number of steps'' would bias the result toward explicit methods.
        Expressing every operation in the same unit is the only way to ask
        ``who is cheaper at the same accuracy''.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Implicit Euler loses on both sides of the product}
        \footnotesize
        \begin{itemize}
          \setlength{\itemsep}{0.05em}
          \item Implicit Euler is $1$st order: to hit an error of $10^{-6}$
                it needs a small $h$, so the number of steps $40/h$ stays up
          \item Each step still costs $3$ units, three times explicit Euler
          \item Multiply the two and at $10^{-6}$ IE costs
                \alert{$3.66\times$} as much as EE
        \end{itemize}
      \end{alertblock}
      \begin{block}{Why RK4 wins at the end}
        \footnotesize
        RK4 costs $4$ units per step (more than IE), but it is $4$th order
        and its step-size limit is also larger ($8.209\times10^{-4}$ vs the
        small $h$ forced on IE by accuracy).  At $10^{-7}$ RK4 reaches the
        target at its cheapest stable step, while the others would need even
        smaller $h$ and still higher cost.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion 4 (top): the Newton tolerance has an upper bound}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \includegraphics[width=\linewidth]{F10_newton_tolerance_sweep.pdf}
      \begin{center}\footnotesize \fig{F10}: conservation defect (flat) vs solution error (not flat)\end{center}
      \vspace{0.3em}
      \footnotesize
      \textbf{Basics}: implicit Euler must solve $G(u)=u-y_n-hf(u)=0$ each
      step.  The standard predictor is $u^0=y_n+hf(y_n)$, which is exactly
      one explicit-Euler step.  If the residual $\|G(u^0)\|\le\text{tol}$,
      the predictor is accepted and \alert{Newton takes zero updates}.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Why a small $h$ makes the residual small automatically}
        \footnotesize
        Substituting the predictor gives
        $G(u^0)=-\frac{h^2}{2}J_f f+O(h^3)$, i.e.\ the residual is
        $O(h^2)$ (about $10^{-6}$ when $h=10^{-3}$).  So there is a threshold
        \[
          h^\ast=\sqrt{\text{tol}\big/\|J_f f\|_\infty}
        \]
        Using $\|J_f f\|_\infty=8.2324\times10^{-3}$:
      \end{block}
      {\footnotesize
      \setlength{\tabcolsep}{5pt}%
      \begin{tabular}{@{}lrr@{}}
        \toprule
        tolerance & error inflation & $h^\ast$ \\
        \midrule
        $10^{-6}$          & $2.67$          & $1.1\times10^{-2}$ \\
        $\mathbf{10^{-8}}$ & $\mathbf{0.88}$ & $1.1\times10^{-3}$ \\
        $10^{-10}$         & $1.00$          & $1.1\times10^{-4}$ \\
        \bottomrule
      \end{tabular}}
      \vspace{0.2em}

      \alert{$10^{-8}$ suffices}; tighter is waste.  But $10^{-6}$ pushes
      $h^\ast$ past the explicit stability bound $5.9\times10^{-4}$ by
      \alert{nineteen times}, and \emph{the error estimator sees nothing}.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion 4, why?  Loosening the tolerance changes the method}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      {\footnotesize
      \setlength{\tabcolsep}{5pt}%
      \begin{tabular}{@{}lrr@{}}
        \toprule
        tolerance & error inflation & zero-update fraction \\
        \midrule
        $10^{-2}$  & $1.96\times10^{7}$ & 60.0\% \\
        $10^{-6}$  & $\mathbf{2.67}$    & 70.3\% \\
        $10^{-8}$  & $0.88$             & 68.8\% \\
        $10^{-10}$ & $1.00$             & 18.4\% \\
        \bottomrule
      \end{tabular}}
      \vspace{0.35em}

      \footnotesize
      The third column is the fraction of steps whose predictor was accepted
      without any Newton update.  $10^{-8}$ and $10^{-6}$ look similar
      ($68.8\%$ vs $70.3\%$), but the error inflation differs by a factor of
      $3$ --- \alert{this fraction alone does not diagnose the problem}.

      \vspace{0.45em}
      \begin{block}{An independent second test (\tab{T18})}
        \footnotesize
        At a fixed state, trial by trial:
        tol $=10^{-14}$ gives \alert{$0/7$} purely explicit steps;
        tol $=10^{-6}$ gives \alert{$7/7$} purely explicit steps ---
        the method is genuinely changed.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Why the error estimator does not notice}
        \footnotesize
        Adaptive / dense-output estimators use
        $E=\|y_{\text{whole}}-y_{\text{two halves}}\|$.
        This is a \emph{difference between two sides}; as long as both sides
        use the same scheme it gives the expected $O(h^2)$ signal and the
        controller is happy.
        \begin{itemize}
          \setlength{\itemsep}{0.05em}
          \item Fully explicit and fully Newton-corrected runs at the same $h$
                give $E$ values within $1\%$ of each other
          \item The fitted $h$-exponent stays inside $[1.89,2.10]$ --- no
                anomaly
        \end{itemize}
      \end{alertblock}
      \begin{block}{The take-away: a silent method switch}
        \footnotesize
        The tolerance is not ``the tighter the better''; it must be tight
        enough that Newton is \alert{actually working}.  Here $10^{-8}$ is
        right.  The right test is the computable $h^\ast$, not the error
        estimator after the fact.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[shrink=2]{Conclusion 4 (bottom): adaptivity is stability-limited, not accuracy-limited}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \includegraphics[width=0.78\linewidth]{F11_adaptive_stepping.pdf}
      \begin{center}\footnotesize \fig{F11}: $h(t)$, err$/$tol trace, and a failing norm\end{center}
      \vspace{0.2em}
      {\footnotesize
      \textbf{Basics}: adaptivity compares one whole step with two half steps,
      then scales $h$ to meet tol.  Accepted steps use \alert{two half steps},

      \begin{tabular}{@{}lrr@{}}

      \vspace{0.25em}
      At looser tolerances $h/2$ sits $6$--$7\%$ above the frozen bound while
      err$/$tol is only $0.17$--$0.31$:
      \alert{the accuracy budget is nowhere near spent}.
    \end{column}
    \begin{column}{0.51\textwidth}
      \begin{alertblock}{Another trap: the absolute $L^\infty$ controller is wrong}
        \footnotesize
        $y_2$ is $O(10^{-5})$ but the error scale is $10^{-10}$.  With an
        absolute tolerance $10^{-6}$ the controller accepts steps far beyond
        the RK4 stability limit:
        \begin{itemize}
          \setlength{\itemsep}{0.05em}
          \item tol $=10^{-6}$: runs to the end, but $y_2$ reaches
                \alert{$-4.77\times10^{-5}$}, total error $7.83\times10^{-3}$
          \item tol $=10^{-4}$: \alert{aborts} at $t=1.94$ with
                $\min y=-9.03\times10^{2}$
        \end{itemize}
      \end{alertblock}
      \begin{block}{The fix}
        \footnotesize
        Use a \alert{scaled (mixed) norm}:
        $\max_i |e_i|\big/(\mathrm{atol}_i+|y_i|)\le\mathrm{tol}$.
        \tab{T12} reports both norms.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{One-sentence summary}
  \begin{block}{Root of the difficulty: two clocks, and the fast one pins everything}
    The two clocks start only $1.7\times$ apart, but by $t=40$ they differ by
    \alert{$1.58\times10^{5}\times$}.  The fast one pins the explicit step size
    at $h<5.9\times10^{-4}$, so crossing $[0,40]$ needs at least
    $6.8\times10^{4}$ steps --- \emph{and this restriction does not relax even
    after the fast process has died down}.  That is stiffness.
  \end{block}

  \begin{alertblock}{Methodological lesson: validation is easier to fool than solving}
    The conservation defect is $1.1\times10^{-13}$ (looks perfect), yet the same
    run has concentration $-921.8$ and error $1.08\times10^{10}$.  ``Did not
    blow up'' and ``passed a check'' are \alert{not the same as ``right''}.
    The reason is that conservation is \emph{exactly} preserved by the
    discretisation, so it has no discriminating power; only an independently
    built ruler (three confirmation routes agreeing to $10.76$ digits) compared
    point by point counts.
  \end{alertblock}

  \begin{exampleblock}{``More stable'' and ``cheaper'' are two different things}
    ``Implicit is more stable'' is true; ``implicit is cheaper'' is
    \alert{not true} in the reachable accuracy range --- at $10^{-6}$ error EE
    costs $3.0\times10^{5}$ cost units while IE costs $1.1\times10^{6}$.
    You only see this by putting \emph{cost, error and stability bound} on the
    same plot.
  \end{exampleblock}
\end{frame}
